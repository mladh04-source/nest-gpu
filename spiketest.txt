import nestgpu

nestgpu.SetTimeResolution(0.1)
nestgpu.SetRandomSeed(12345)
nestgpu.SetVerbosityLevel(0)

N = 200
neurons = nestgpu.Create("iaf_psc_exp", N)

# Enable spike counting 
nestgpu.ActivateSpikeCount(neurons)

nestgpu.Simulate(100.0)

i0 = int(neurons[0])   # first neuron id
#  First retrieve all possible status keys 
scal_vars = nestgpu.GetScalVarNames(i0)      
int_vars  = nestgpu.GetIntVarNames(i0)       
params    = nestgpu.GetScalParamNames(i0)    

print("ScalVars:", scal_vars)
print("IntVars :", int_vars)
print("Params  :", params)

#  Search for spike count candidates 
cands = [v for v in int_vars if "spike" in v.lower() and "count" in v.lower()]
print("SpikeCount candidates:", cands)

if not cands:
    raise RuntimeError("No spike_count-like IntVar found. See IntVars above.")

k = cands[0]

#  Retrieve SpikeCount
counts = nestgpu.GetNeuronStatus(neurons, k)

# output
flat = []
for c in counts:
    if isinstance(c, list):
        flat.extend(c)
    else:
        flat.append(c)

total = sum(flat)

print("Using var:", k)
print("TOTAL spikes:", total)
print("First 10 counts:", flat[:10])
